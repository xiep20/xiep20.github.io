obj/fbx转gltf和gltf压缩 ​

详情

Go 3D 模型格式转换工具

概述 ​

这是一个使用 Go 语言开发的 3D 模型格式转换工具，支持将 OBJ、FBX 等常见 3D 模型格式转换为 GLTF 格式，并支持 Draco 压缩优化。

主要功能 ​

  • 格式转换：支持 OBJ → GLTF、FBX → GLTF 格式转换
  • 数据提取：从源格式中提取顶点、索引、材质等几何数据
  • GLTF 生成：生成标准的 GLTF 文档结构
  • 压缩优化：支持 Draco 几何压缩，减少文件大小
  • 批量处理：支持命令行批量转换多个文件

依赖库 ​

go
import (
    "github.com/qmuntal/gltf"    // GLTF 文件读写
    "github.com/udhos/gwob"       // OBJ 文件解析
    "github.com/mogaal/gofbx"     // FBX 文件解析
)

使用方法 ​

bash
# OBJ 转 GLTF
go run converter.go model.obj model.gltf

# FBX 转 GLTF
go run converter.go model.fbx model.gltf

# 转换并压缩
go run converter.go model.obj model.gltf --compress

技术原理 ​

转换流程 ​
  1. 文件解析：使用专门的库解析源格式文件
  2. 数据提取：提取几何数据（顶点、索引、UV、法线等）
  3. 格式转换：将数据转换为 GLTF 标准结构
  4. 数据优化：应用 Draco 压缩减少文件大小
  5. 文件输出：生成最终的 GLTF 文件

注意事项 ​

  • 文件格式：目前支持 OBJ 和 FBX 格式输入
  • 数据完整性：转换过程中可能丢失部分高级特性（如动画、骨骼等）
  • 性能考虑：大文件转换可能需要较长时间
  • 内存使用：转换过程中会占用较多内存
  • 依赖管理：需要正确安装和配置相关 Go 模块

// OBJ 转 GLTF
func objToGLTF(input, output string) {
    // 解析OBJ
    obj, err := gwob.NewObjFromFile(input, &gwob.ObjParserOptions{})
    if err != nil {
        log.Fatal("解析OBJ文件失败:", err)
    }

    doc := gltf.NewDocument()

    // 填充顶点数据
    buffer := gltf.NewBuffer()
    buffer.Data = obj.CoordsToFloat32() // 假设顶点数据转换为float32
    doc.Buffers = append(doc.Buffers, buffer)

    bufferView := gltf.NewBufferView()
    bufferView.Buffer = 0
    bufferView.ByteLength = len(buffer.Data)
    doc.BufferViews = append(doc.BufferViews, bufferView)

    accessor := gltf.NewAccessor()
    accessor.BufferView = gltf.Index(0)
    accessor.ComponentType = gltf.ComponentFloat
    accessor.Type = gltf.AccessorVec3
    accessor.Count = uint32(len(obj.Vertices) / 3
    doc.Accessors = append(doc.Accessors, accessor)

    // 创建Mesh
    primitive := gltf.NewPrimitive()
    primitive.Attributes["POSITION"] = 0
    mesh := gltf.NewMesh()
    mesh.Primitives = []*gltf.Primitive{primitive}
    doc.Meshes = append(doc.Meshes, mesh)

    // 导出GLTF
    if err := gltf.Save(doc, output); err != nil {
        log.Fatal("导出GLTF文件失败:", err)
    }
}

// GLTF 压缩
func compressGLTF(input, output string) {
    // 读取GLTF文件
    doc, err := gltf.Open(input)
    if err != nil {
        log.Fatal("读取GLTF文件失败:", err)
    }

    // 压缩几何数据
    for _, buffer := range doc.Buffers {
        buffer.Data = compressWithDraco(buffer.Data)
    }

    // 保存压缩后的GLTF
    if err := gltf.Save(doc, output); err != nil {
        log.Fatal("保存压缩GLTF文件失败:", err)
    }
}

func main() {
    if len(os.Args) < 3 {
        log.Fatal("用法: converter <输入文件> <输出文件> [--compress]")
    }
    input := os.Args[1]
    output := os.Args[2]
    compress := len(os.Args) > 3 && os.Args[3] == "--compress"

    ext := strings.ToLower(filepath.Ext(input))
    switch ext {
    case ".obj":
        objToGLTF(input, output)
    case ".fbx":
        fbxToGLTF(input, output)
    default:
        log.Fatal("不支持的文件格式")
    }

    if compress {
        // 实现Draco压缩逻辑
        compressedOutput := strings.Replace(output, ".gltf", "-compressed.gltf", 1)
        compressGLTF(output, compressedOutput)
    }
}
