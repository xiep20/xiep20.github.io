Web Worker多线程加密地图瓦片解密方法

一个利用Web Worker多线程进行加密地图瓦片的解密方法

主线程代码 ​

js
import Worker from './img.worker.js'

let worker = new Worker()
// 向子线程发送消息  
worker.postMessage({img:imgurl,tileId: tile.id})
// 注册监听函数，接收子线程消息
worker.onmessage = (event) => {
    if(document.querySelector('#'+event.data.tileId)){
        document.querySelector('#'+event.data.tileId).src = event.data.img
        // tile.src = event.data.img
    }
}

// 监听 Worker 的错误信息
worker.onerror = function(error) {
    console.error('Worker 发生错误：', error);
};

Worker 线程代码 (img.worker.js) ​

js
import axios from "axios";
import CryptoJS from 'crypto-js'

// 子线程注册监听函数，接收到主线程消息后，将消息回发
self.addEventListener('message', (e) => {
    let imgurl = e.data.img
    let tileId = e.data.tileId
    console.log(imgurl)
    
    axios({
        method: 'get',
        url: imgurl,
        responseType: 'arraybuffer',
        headers: {Accept: 'application/octet-stream'}
    }).then((response) => {
        // 处理响应数据
        if (response && response.data) {
            let imgbase = uaprocess(response.data, 'ABCDEF1234567890', 'ABCDEF1234567890')
            self.postMessage({img:"data:image/gif;base64," + imgbase,tileId: tileId});
        }
    }).catch(function (error) {
        if(error.response.status !== 404){
            debugger
        }
        console.log(error);
    });
});

function uaprocess(dcBase64String, key, iv) {
    let keyWA = CryptoJS.enc.Utf8.parse(key)
    let ivWA = CryptoJS.enc.Utf8.parse(iv)

    // 将二进制数据转换为Base64字符串
    dcBase64String = arrayBufferToBase64(dcBase64String)

    // 解密
    let decryptedData = CryptoJS.AES.decrypt(dcBase64String, keyWA, {
        iv: ivWA,
        mode: CryptoJS.mode.CBC,
        padding: CryptoJS.pad.Pkcs7
    })

    // 将解密结果转换为 Base64 字符串
    var d64 = decryptedData.toString(CryptoJS.enc.Base64)
    return d64;
}

function arrayBufferToBase64(buffer) {
    var binary = '';
    var bytes = new Uint8Array(buffer);
    var len = bytes.byteLength;
    for (var i = 0; i < len; i++) {
        binary += String.fromCharCode(bytes[i]);
    }
    // 对二进制字符串进行Base64编码
    return btoa(binary);
}

主要特点 ​

  • 使用 Web Worker 在后台线程处理加密瓦片的解密
  • 避免主线程阻塞，提升用户体验
  • 支持 AES 解密和 Base64 编码转换
  • 通过消息传递机制与主线程通信

完整代码实现 ​

js
import mqtt from "mqtt/dist/mqtt.min";

export default class mqttModel {
    constructor(_connection, callback = {}) {
        this.connection = {
            protocol: "ws",
            host: "broker.emqx.io",
            // ws: 8083; wss: 8084
            port: 8083,
            endpoint: "/mqtt",
            // for more options, please refer to https://github.com/mqttjs/MQTT.js#mqttclientstreambuilder-options
            clean: true,
            connectTimeout: 30 * 1000, // ms
            reconnectPeriod: 4000, // ms
            clientId: "emqx_vue_" + Math.random().toString(16).substring(2, 8),
            // auth
            username: "emqx_test",
            password: "emqx_test",
        };
        this.callback = callback
        this.connection = Object.assign(this.connection, _connection)

        this.receiveNews = "";
        this.qosList = [0, 1, 2];
        this.client = {
            connected: false,
        };
        this.retryTimes = 0;
        this.connecting = false;
        this.subscribeSuccess = false;
        this.createConnection();
    }

    initData() {
        this.client = {
            connected: false,
        };
        this.retryTimes = 0;
        this.connecting = false;
        this.subscribeSuccess = false;
    }

    handleOnReConnect() {
        this.retryTimes += 1;
        if (this.retryTimes > 5) {
            try {
                this.client.end();
                this.initData();
                console.log("Connection maxReconnectTimes limit, stop retry");
            } catch (error) {
                console.log(error.toString());
            }
        }
    }

    createConnection() {
        try {
            this.connecting = true;
            const { protocol, host, port, endpoint, ...options } = this.connection;
            const connectUrl = `${protocol}://${host}:${port}${endpoint}`;
            this.client = mqtt.connect(connectUrl, options);
            
            if (this.client.on) {
                this.client.on("connect", () => {
                    this.connecting = false;
                    console.log("Connection succeeded!");
                    if (this.callback["connect"] && typeof this.callback["connect"] === 'function') {
                        this.callback["connect"]()
                    }
                });
                
                this.client.on("reconnect", this.handleOnReConnect);
                
                this.client.on("error", (error) => {
                    console.log("Connection failed", error);
                    if (this.callback["error"] && typeof this.callback["error"] === 'function') {
                        this.callback["error"](error)
                    }
                });
                
                this.client.on("message", (topic, message) => {
                    if (this.callback["message"] && typeof this.callback["message"] === 'function') {
                        this.callback["message"](topic, message)
                    }
                });
            }
        } catch (error) {
            this.connecting = false;
            console.log("mqtt.connect error", error);
        }
    }

    // 订阅主题
    doSubscribe(subscription) {
        // subscription: {
        //     topic: "topic/mqttx",
        //     qos: 0,
        // },
        const { topic, qos } = subscription
        this.client.subscribe(topic, { qos }, (error, res) => {
            if (error) {
                console.log('Subscribe to topics error', error)
                return
            }
            this.subscribeSuccess = true
            console.log('Subscribe to topics res', res)
        })
    }

    // 取消订阅
    doUnSubscribe(subscription) {
        const { topic } = subscription
        this.client.unsubscribe(topic, error => {
            if (error) {
                console.log('Unsubscribe error', error)
            }
        })
    }

    // 消息发布
    doPublish(publish) {
        // publish: {
        //     topic: "topic/browser",
        //     qos: 0,
        //     payload: '{ "msg": "Hello, I am browser." }',
        // },
        const { topic, qos, payload } = publish
        this.client.publish(topic, payload, { qos }, error => {
            if (error) {
                console.log('Publish error', error)
            }
        })
    }

    // 断开连接
    destroyConnection() {
        if (this.client.connected) {
            try {
                this.client.end(false, () => {
                    this.initData()
                    console.log('Successfully disconnected!')
                })
            } catch (error) {
                console.log('Disconnect failed', error.toString())
            }
        }
    }
}

使用方法 ​

js
// 创建MQTT实例
const mqttClient = new mqttModel({
    host: 'your-mqtt-broker.com',
    port: 8083,
    username: 'your-username',
    password: 'your-password'
}, {
    connect: () => {
        console.log('MQTT连接成功');
        // 订阅主题
        mqttClient.doSubscribe({
            topic: 'test/topic',
            qos: 0
        });
    },
    message: (topic, message) => {
        console.log('收到消息:', topic, message.toString());
    },
    error: (error) => {
        console.error('MQTT错误:', error);
    }
});

// 发布消息
mqttClient.doPublish({
    topic: 'test/topic',
    qos: 0,
    payload: 'Hello MQTT!'
});

// 断开连接
mqttClient.destroyConnection();
