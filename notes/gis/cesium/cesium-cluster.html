点聚合效果（cluster） ​

使用cesium本身的dataSource.clustering，不适用大量数据

js
import axios from "axios";
export default class clustyPoi {
    constructor(_viewer, opts = {}) {
        this.viewer = _viewer;
        this.options = {
            pixelRange: 45,
            enabled: true,
            minimumClusterSize: 3,
            img: "",
            width: 32,
            height: 32,
            colorArr: [{
                num: 1,
                size: 36,
                color: "#1c86d1"
            },
            {
                num: 50,
                size: 38,
                color: "#67c23a"
            },
            {
                num: 100,
                size: 42,
                color: "#e6a23c"
            }, {
                num: 300,
                size: 44,
                color: "#f56c6c"
            }],
            accuracy: function (num) {
                var text = num;
                if (num > 10 && num < 100) {
                    text = parseInt(num / 10) * 10 + "+"
                } else if (num > 100 && num < 1000) {
                    text = parseInt(num / 100) * 100 + "+"
                } else if (num == 1000) {
                    text = "1k"
                } else if (num > 1000) {
                    text = parseInt(num / 1000) + "k+"
                }
                return text;
            },
        };
        this.clustericon = {}
        this.options = Object.assign(this.options, opts);
        this.clusterDataSource = null;
    }
    async clusterPoi(url) {
        let that = this
        // 先清除上一次聚合
        this.clearPoiCluster()

        let res = await axios.get(url);
        this.POIdata = res.data
        var dataSourceForCluster = new Cesium.CustomDataSource('cluster');
        for (let i = 0; i < res.data.length; i++) {
            var obj = res.data[i];
            let xy = obj.geom.split("(")[1].split(")")[0].split(" ");


            dataSourceForCluster.entities.add({
                position: new Cesium.Cartesian3.fromDegrees(Number(xy[0]), Number(xy[1])),
                billboard: {
                    image: that.options.img,
                    width: that.options.width,
                    height: that.options.height,
                },
                properties: new Cesium.PropertyBag({
                    type: "point",
                    details: obj
                })
            });
        }
        let dataSourcePromise = viewer.dataSources.add(dataSourceForCluster);

        // 添加聚合数据源
        this.addClusterDataSource(this.viewer, dataSourcePromise);
    }
    /**
  * 添加聚合数据源
  */
    addClusterDataSource(viewer, dataSourcePromise) {
        let that = this
        dataSourcePromise.then(function (dataSource) {
            that.clusterDataSource = dataSource;


            dataSource.clustering.enabled = that.options.enabled; //是否聚合
            dataSource.clustering.pixelRange = that.options.pixelRange; //聚合范围（单位px）
            dataSource.clustering.minimumClusterSize = that.options.minimumClusterSize; //最小屏幕聚合对象数值（小于等于该数值，不聚合）

            var removeListener;
            //聚合
            function customStyle() {
                if (Cesium.defined(removeListener)) {
                    removeListener();
                    removeListener = undefined;
                } else {
                    removeListener = dataSource.clustering.clusterEvent.addEventListener(
                        function (clusteredEntities, cluster) {
                            console.log(clusteredEntities.length)
                            // console.log(that, clusteredEntities, cluster)// 关闭自带的显示聚合数量的标签
                            cluster.label.show = false;
                            cluster.billboard.show = true;
                            cluster.billboard.id = cluster.label.id;
                            cluster.billboard.verticalOrigin = Cesium.VerticalOrigin.BOTTOM;

                            var xx = -1;
                            for (var i = 0; i < that.options.colorArr.length; i++) {
                                if (clusteredEntities.length > that.options.colorArr[i].num) {
                                    xx = i;
                                }
                            }
                            if (xx == -1) {
                                cluster.billboard.image = that.options.img
                            } else {
                                cluster.billboard.image = that.drawImage(clusteredEntities.length, that.options.colorArr[xx]
                                    .size, that.options.colorArr[xx].color);
                            }
                        }
                    );
                }
            }
            customStyle();
        });
    }
    drawImage(text, size, color) {
        text = this.options.accuracy(text);
        if (this.clustericon[text + "_" + size + "_" + color]) {
            return this.clustericon[text + "_" + size + "_" + color];
        }
        var canvas = document.createElement("canvas");
        canvas.height = size;
        canvas.width = size;
        var ctx = canvas.getContext('2d');
        // //画圈
        // ctx.beginPath();
        // ctx.arc(size / 2, size / 2, size / 2, 0, Math.PI * 2, true);
        // ctx.fillStyle = color+"88";
        // ctx.closePath();
        // ctx.fill();
        //画圈
        ctx.beginPath();
        ctx.arc(size / 2, size / 2, size / 2, 0, Math.PI * 2, true);

        var r1 = size / 2.8;    // 第一个圆的半径
        var r2 = size / 2.3;   // 第二个圆的半径
        var r3 = size / 2;   // 第三个圆的半径
        var radialGradient1 =
            ctx.createRadialGradient(size / 2, size / 2, r1, size / 2, size / 2, r2, size / 2, size / 2,r3);
        radialGradient1.addColorStop(0, color + "cc");
        radialGradient1.addColorStop(1, color + "88");
        radialGradient1.addColorStop(1, color + "33");
        ctx.fillStyle = radialGradient1;
        ctx.closePath();
        ctx.fill();

        ctx.font = '16px bold 楷体';
        ctx.fillStyle = '#ffffff';
        var tx = (size - (text + "").length * 9) / 2;
        ctx.fillText(text, tx, (size / 2 + 6));
        var canvasdataurl = canvas.toDataURL();
        this.clustericon[text + "_" + size + "_" + color] = canvasdataurl;
        return canvasdataurl;
    }
    /**
    * 清除poi聚合效果
    */
    clearPoiCluster() {
        // 清除billboard
        if (this.clusterDataSource) {
            this.viewer.dataSources.remove(this.clusterDataSource)
            this.clusterDataSource = null
        }
    }
}
