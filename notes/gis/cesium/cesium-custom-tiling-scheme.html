自定义切片方案类 ​

针对cesium加载自定义切片方案类，一个坐标系的自定义切片方案。

CustomGeographicTilingScheme.js 与 scheme.js。

  1. CustomGeographicTilingScheme.js
  • 继承自 Cesium.GeographicTilingScheme，是对 Cesium 原生地理切片方案的扩展。
  • 支持自定义原点、层级偏移、瓦片分辨率数组等高级功能，适用于需要灵活控制切片分布和分辨率的场景。
  • 主要用于自定义地图服务的切片坐标与地理坐标的转换。
  1. GeographicTilingScheme2.js
  • 自定义实现的地理切片方案，未继承 Cesium 的类，而是完全自定义实现。
  • 提供了基本的切片分块、坐标转换等功能，适合标准的地理切片需求。
  • 主要用于简单的地理切片坐标与地理坐标的转换。

二、主要属性与构造参数

属性/参数CustomGeographicTilingScheme.jsGeographicTilingScheme2.js
ellipsoid继承自父类，支持自定义支持自定义
rectangle继承自父类，支持自定义支持自定义
projection继承自父类支持自定义
origin支持自定义原点（如[-180, 90]）不支持
zoomOffset支持层级偏移不支持
tileSize支持自定义瓦片像素大小不支持
resolutions支持每级分辨率数组不支持
numberOfLevelZeroTilesX/Y继承自父类，支持自定义支持自定义

三、主要方法对比

方法名CustomGeographicTilingScheme.jsGeographicTilingScheme2.js
tileXYToRectangle支持自定义原点、分辨率、偏移，适合多种坐标系标准实现，按等分方式分块
positionToTileXY支持自定义原点、分辨率、偏移标准实现，按等分方式分块
getNumberOfXTilesAtLevel / getNumberOfYTilesAtLevel继承自父类自定义实现，按2的幂次递增
rectangleToNativeRectangle / tileXYToNativeRectangle无有，支持弧度与度的转换

四、实现细节差异

  1. 分辨率与原点
  • CustomGeographicTilingScheme 支持通过 resolutions 数组和 origin 精确控制每一级的分辨率和切片起点，适合非标准切片体系（如部分国产地图服务）。
  • GeographicTilingScheme2 采用标准的等分切片方式，原点和分辨率不可灵活配置。
  1. 层级偏移
  • CustomGeographicTilingScheme 支持 zoomOffset，可灵活适配不同服务的层级定义。
  • GeographicTilingScheme2 不支持层级偏移。

五、适用场景建议

  • CustomGeographicTilingScheme.js
    适用于需要高度自定义切片体系（如自定义原点、分辨率、层级偏移）的场景，推荐用于对接国产地图服务、特殊投影或自定义瓦片服务。

  • GeographicTilingScheme2.js (GeographicTilingScheme2)
    实现简单，适用于标准地理切片体系，需求简单、无需自定义分辨率和原点的场景。

js
// GeographicTilingScheme2.js：自定义地理切片方案，用于地图瓦片的分块与坐标转换
function GeographicTilingScheme2 (options) {
  debugger // 调试用断点
  // 初始化参数，若未传入则使用默认值
  options = Cesium.defaultValue(options, Cesium.defaultValue.EMPTY_OBJECT)
  // 椭球体，默认为WGS84
  this._ellipsoid = Cesium.defaultValue(options.ellipsoid, Cesium.Ellipsoid.WGS84)
  // 切片范围，默认为最大矩形范围
  this._rectangle = Cesium.defaultValue(options.rectangle, Cesium.Rectangle.MAX_VALUE)
  // 投影方式，使用地理投影
  this._projection = new Cesium.GeographicProjection(this._ellipsoid)
  // 零级切片的X方向数量，默认为2
  this._numberOfLevelZeroTilesX = Cesium.defaultValue(
    options.numberOfLevelZeroTilesX,
    2
  )
  // 零级切片的Y方向数量，默认为1
  this._numberOfLevelZeroTilesY = Cesium.defaultValue(
    options.numberOfLevelZeroTilesY,
    1
  )
}

// 定义只读属性，便于外部访问
Object.defineProperties(GeographicTilingScheme2.prototype, {
  ellipsoid: {
    get: function () {
      return this._ellipsoid
    }
  },

  rectangle: {
    get: function () {
      return this._rectangle
    }
  },

  projection: {
    get: function () {
      return this._projection
    }
  }
})

// 获取指定层级下X方向的切片数量
GeographicTilingScheme2.prototype.getNumberOfXTilesAtLevel = function (level) {
  return this._numberOfLevelZeroTilesX << level // 位运算，等价于*2^level
}

// 获取指定层级下Y方向的切片数量
GeographicTilingScheme2.prototype.getNumberOfYTilesAtLevel = function (level) {
  return this._numberOfLevelZeroTilesY << level // 位运算，等价于*2^level
}

// 将弧度制矩形转换为度制矩形
GeographicTilingScheme2.prototype.rectangleToNativeRectangle = function (
  rectangle,
  result
) {
  // 检查rectangle参数是否定义
  Cesium.Check.defined('rectangle', rectangle)

  // 弧度转度
  const west = Cesium.Math.toDegrees(rectangle.west)
  const south = Cesium.Math.toDegrees(rectangle.south)
  const east = Cesium.Math.toDegrees(rectangle.east)
  const north = Cesium.Math.toDegrees(rectangle.north)

  // 若未传入result，则新建一个Rectangle对象
  if (!Cesium.defined(result)) {
    return new Cesium.Rectangle(west, south, east, north)
  }

  // 赋值到result对象
  result.west = west
  result.south = south
  result.east = east
  result.north = north
  return result
}

// 根据切片坐标获取度制矩形范围
GeographicTilingScheme2.prototype.tileXYToNativeRectangle = function (
  x,
  y,
  level,
  result
) {
  // 先获取弧度制矩形，再转为度制
  const rectangleRadians = this.tileXYToRectangle(x, y, level, result)
  rectangleRadians.west = Cesium.Math.toDegrees(rectangleRadians.west)
  rectangleRadians.south = Cesium.Math.toDegrees(rectangleRadians.south)
  rectangleRadians.east = Cesium.Math.toDegrees(rectangleRadians.east)
  rectangleRadians.north = Cesium.Math.toDegrees(rectangleRadians.north)
  return rectangleRadians
}

// 根据切片坐标获取弧度制矩形范围
GeographicTilingScheme2.prototype.tileXYToRectangle = function (
  x,
  y,
  level,
  result
) {
  const rectangle = this._rectangle

  // 计算当前层级的切片数量
  const xTiles = this.getNumberOfXTilesAtLevel(level)
  const yTiles = this.getNumberOfYTilesAtLevel(level)

  // 计算每个切片的宽度和经度范围
  const xTileWidth = rectangle.width / xTiles
  const west = x * xTileWidth + rectangle.west
  const east = (x + 1) * xTileWidth + rectangle.west

  // 计算每个切片的高度和纬度范围
  const yTileHeight = rectangle.height / yTiles
  const north = rectangle.north - y * yTileHeight
  const south = rectangle.north - (y + 1) * yTileHeight

  // 若未传入result，则新建一个Rectangle对象
  if (!Cesium.defined(result)) {
    result = new Cesium.Rectangle(west, south, east, north)
  }

  // 赋值到result对象
  result.west = west
  result.south = south
  result.east = east
  result.north = north
  return result
}

// 根据经纬度坐标获取所在切片的XY索引
GeographicTilingScheme2.prototype.positionToTileXY = function (
  position,
  level,
  result
) {
  const rectangle = this._rectangle
  // 判断点是否在切片范围内
  if (!Cesium.Rectangle.contains(rectangle, position)) {
    // 若不在范围内，返回undefined
    return undefined
  }

  // 计算当前层级的切片数量
  const xTiles = this.getNumberOfXTilesAtLevel(level)
  const yTiles = this.getNumberOfYTilesAtLevel(level)

  // 计算每个切片的宽度和高度
  const xTileWidth = rectangle.width / xTiles
  const yTileHeight = rectangle.height / yTiles

  let longitude = position.longitude
  // 处理跨越180度经线的情况
  if (rectangle.east < rectangle.west) {
    longitude += Cesium.Math.TWO_PI
  }

  // 计算X方向的切片索引
  let xTileCoordinate = ((longitude - rectangle.west) / xTileWidth) | 0
  if (xTileCoordinate >= xTiles) {
    xTileCoordinate = xTiles - 1
  }

  // 计算Y方向的切片索引
  let yTileCoordinate =
    ((rectangle.north - position.latitude) / yTileHeight) | 0
  if (yTileCoordinate >= yTiles) {
    yTileCoordinate = yTiles - 1
  }

  // 若未传入result，则新建一个Cartesian2对象
  if (!Cesium.defined(result)) {
    return new Cesium.Cartesian2(xTileCoordinate, yTileCoordinate)
  }

  // 赋值到result对象
  result.x = xTileCoordinate
  result.y = yTileCoordinate
  return result
}

// 导出自定义切片方案类
export default GeographicTilingScheme2
js
/*
 * @FilePath: \map_scene\src\utils\scheme\CustomGeographicTilingScheme.js
 * @Author: xieping
 * @Date: 2024-06-21 09:53:47
 * @LastEditTime: 2024-07-18 09:27:39
 * @LastEditors: xieping
 * @Description: 自定义的地图瓦片投影方案
 * 支持传入原点、层级偏移、瓦片分辨率
 */
class CustomGeographicTilingScheme extends Cesium.GeographicTilingScheme {
  constructor (options = {}) {
    super(options)
    this._origin = options.origin || [-180, 90]
    this._zoomOffset = options.zoomOffset || 0
    this._tileSize = options.tileSize || 256
    this._resolutions = options.resolutions || []
  }

  /**
 * 获取当前缩放偏移量。
 *
 * @returns {number} 当前的缩放偏移量。
 */
  get zoomOffset () {
    return this._zoomOffset
  }

  /**
     * 将瓦片坐标(x, y)和层级(level)转换为矩形范围。
     * 此函数用于根据瓦片坐标和层级计算出对应瓦片在地理空间中的矩形范围。
     * @param {Number} x - 瓦片的x坐标。
     * @param {Number} y - 瓦片的y坐标。
     * @param {Number} level - 瓦片的层级。
     * @param {Object} [result] - 可选参数，用于接收计算结果的矩形对象。
     * @returns {Cesium.Rectangle} 表示瓦片范围的矩形对象。如果没有可用的分辨率数据，则返回最大矩形范围。
     */
  tileXYToRectangle (x, y, level, result) {
    // 检查是否存在对应层级的分辨率数据，如果没有则返回最大矩形范围
    if (!this._resolutions || !this._resolutions[level + this._zoomOffset]) {
      return Cesium.Rectangle.MAX_VALUE
    }

    // 计算当前层级的瓦片大小
    const tileRes = this._resolutions[level + this._zoomOffset] * this._tileSize
    // 计算瓦片的西方、南方、东方、北方界限（单位：弧度）
    const west = Cesium.Math.toRadians(this._origin[0] + x * tileRes)
    const south = Cesium.Math.toRadians(this._origin[1] - (y + 1) * tileRes)
    const east = Cesium.Math.toRadians(this._origin[0] + (x + 1) * tileRes)
    const north = Cesium.Math.toRadians(this._origin[1] - y * tileRes)

    // 如果没有提供结果对象，则创建并返回一个新的矩形对象
    if (!Cesium.defined(result)) {
      return new Cesium.Rectangle(west, south, east, north)
    }
    // 如果提供了结果对象，则将其属性设置为计算得到的界限值，并返回该对象
    result.west = west
    result.south = south
    result.east = east
    result.north = north
    return result
  }

  /**
 * 将位置信息转换为瓦片坐标系中的XY坐标。
 * 此函数用于根据给定的位置和地图级别，计算出该位置在瓦片坐标系中的X和Y坐标。
 * 瓦片坐标系是一种将地球表面划分成多个矩形瓦片的系统，每个瓦片用X和Y坐标唯一标识。
 *
 * @param {Cesium.Cartesian3} position 需要转换的位置，使用经度、纬度度量。
 * @param {Number} level 地图的缩放级别。
 * @param {Cesium.Cartesian2} [result] 可选参数，用于存储计算结果的瓦片坐标。如果未提供，则会创建一个新的Cartesian2对象。
 * @returns {Cesium.Cartesian2} 返回包含X和Y坐标的瓦片坐标对象。如果无法进行转换，则返回一个新的空Cartesian2对象。
 */
  positionToTileXY (position, level, result) {
  // 检查是否存在对应级别的分辨率设置，如果没有，则返回空的Cartesian2对象
    if (!this._resolutions || !this._resolutions[level + this._zoomOffset]) {
      return new Cesium.Cartesian2()
    }

    // 计算当前级别的瓦片分辨率
    const tileRes = this._resolutions[level + this._zoomOffset] * this._tileSize
    // 将位置的经度和纬度转换为度
    const longitude = Cesium.Math.toDegrees(position.longitude)
    const latitude = Cesium.Math.toDegrees(position.latitude)

    // 计算瓦片的X坐标
    const xTileCoordinate = Math.floor((longitude - this._origin[0]) / tileRes)
    // 计算瓦片的Y坐标
    const yTileCoordinate = Math.floor((this._origin[1] - latitude) / tileRes)

    // 如果没有提供result参数，创建并返回一个新的Cartesian2对象，包含计算出的瓦片坐标
    if (!Cesium.defined(result)) {
      return new Cesium.Cartesian2(
        Math.max(0, xTileCoordinate),
        Math.max(0, yTileCoordinate)
      )
    }

    // 如果提供了result参数，将其X和Y属性设置为计算出的瓦片坐标
    result.x = xTileCoordinate
    result.y = yTileCoordinate
    // 返回设置好的result参数
    return result
  }
}

export default CustomGeographicTilingScheme
